pymupdf          # PDF text + image extraction  (import as: fitz)

pillow           # Image loading and pre-processing

# --- Vector store ---
chromadb         # Local persistent vector database

# --- Embeddings + LLM (via local Ollama) ---
ollama           # Ollama Python SDK

# --- UI ---
streamlit        # Web-based frontend

# --- File-system monitoring ---
watchdog         # Watch folders for new/modified files

# --- Keyword search fallback ---
rank-bm25        # Fast BM25 keyword search index
